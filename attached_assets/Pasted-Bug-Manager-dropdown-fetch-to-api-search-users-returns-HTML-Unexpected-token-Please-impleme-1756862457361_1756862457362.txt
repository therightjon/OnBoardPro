Example (TypeScript/Express-ish):

app.get('/api/search/users', requireAuth, async (req, res) => {
  try {
    const q = (req.query.q ?? '').toString().trim();
    const role = (req.query.role ?? '').toString().trim(); // empty if not provided

    // Build SQL dynamically but safely (parameterized)
    // Prefer trigram if extension exists; otherwise fallback to ILIKE match.

    const text = `
      WITH params AS (
        SELECT NULLIF($1, '')::text AS q, NULLIF($2, '')::text AS role
      )
      SELECT u.id,
             trim(concat_ws(' ', u.first_name, u.last_name)) AS name,
             CASE
               WHEN (SELECT q FROM params) IS NULL THEN 1.0
               ELSE GREATEST(similarity(lower(trim(concat_ws(' ', u.first_name, u.last_name))), lower((SELECT q FROM params))), 0)
             END AS score
      FROM users u, params p
      WHERE (p.role IS NULL OR u.role = p.role)
        AND (
          p.q IS NULL
          OR trim(concat_ws(' ', u.first_name, u.last_name)) ILIKE '%' || p.q || '%'
          OR similarity(lower(trim(concat_ws(' ', u.first_name, u.last_name))), lower(p.q)) > 0.1
        )
      ORDER BY
        CASE WHEN (SELECT q FROM params) IS NULL THEN 0 ELSE 1 END,
        score DESC,
        name ASC
      LIMIT 20;
    `;

    const { rows } = await db.query(text, [q, role]);
    res.setHeader('Content-Type', 'application/json');
    res.status(200).json({ items: rows.map(r => ({ id: r.id, name: r.name, score: Number(r.score) })), query: q });
  } catch (err) {
    console.error('search users error', err);
    res.status(500).json({ error: 'SEARCH_USERS_FAILED' });
  }
});

export async function searchManagers(q: string) {
  const url = `/api/search/users?q=${encodeURIComponent(q)}&role=manager`;
  const r = await fetch(url, { headers: { 'Accept': 'application/json' } });
  if (!r.ok) {
    const text = await r.text(); // may be HTML
    throw new Error(`searchManagers ${r.status}: ${text.slice(0, 200)}`);
  }
  let json: any;
  try { json = await r.json(); }
  catch (e) {
    const text = await r.text();
    throw new Error(`searchManagers invalid JSON: ${text.slice(0, 200)}`);
  }
  return (json.items ?? []) as { id: string; name: string; score?: number }[];
}
