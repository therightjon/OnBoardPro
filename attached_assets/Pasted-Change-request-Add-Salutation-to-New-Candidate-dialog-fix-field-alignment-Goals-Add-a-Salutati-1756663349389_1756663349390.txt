Database safety (idempotent)
-- Only run if not present
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'salutation_type') THEN
    CREATE TYPE salutation_type AS ENUM ('Mr','Ms','Mrs','Dr','Prof','Other');
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_name='candidates' AND column_name='salutation'
  ) THEN
    ALTER TABLE candidates
    ADD COLUMN salutation salutation_type NOT NULL DEFAULT 'Mr';
  END IF;
END$$;
