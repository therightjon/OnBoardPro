Feature: Template Status dropdown with readiness checks and blocking dialog

Goal
- Replace the static status pill on the Template Detail page with a dropdown control.
- Allowed statuses: Draft, Active, Archived.
- Rule: Active requires at least one active stage on this template.
- If user selects an invalid status, show a dialog that explains what is missing and prevents the change.

RBAC
- Only system_admin and hr_staff can change status.

Database (idempotent)
-- Keep the guard in DB so the API remains correct.
CREATE OR REPLACE FUNCTION template_is_ready(p_template_id uuid)
RETURNS boolean LANGUAGE plpgsql AS $$
DECLARE
  stage_cnt int;
BEGIN
  SELECT COUNT(*) INTO stage_cnt
  FROM template_stages
  WHERE template_id = p_template_id AND is_active = true;
  RETURN stage_cnt > 0;
END$$;

CREATE OR REPLACE FUNCTION enforce_template_ready()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.is_active = true AND NOT template_is_ready(NEW.id) THEN
    RAISE EXCEPTION 'Template cannot be set to Active until it has at least one stage'
      USING ERRCODE = 'check_violation';
  END IF;
  RETURN NEW;
END$$;

DROP TRIGGER IF EXISTS trg_templates_guard_active ON templates;
CREATE CONSTRAINT TRIGGER trg_templates_guard_active
AFTER INSERT OR UPDATE OF is_active ON templates
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION enforce_template_ready();

-- Optional: a helper view for readiness details
CREATE OR REPLACE VIEW template_readiness AS
SELECT
  t.id AS template_id,
  COUNT(ts.*) FILTER (WHERE ts.is_active) AS active_stage_count
FROM templates t
LEFT JOIN template_stages ts ON ts.template_id = t.id
GROUP BY t.id;

API
1) GET /api/templates/:id/readiness
   Returns readiness info for UI checks.
   SELECT active_stage_count FROM template_readiness WHERE template_id = $1;

2) PATCH /api/templates/:id/status
   Body: { status: 'draft' | 'active' | 'archived' }
   Behavior:
     - Map to columns:
       - draft -> is_active=false, archived=false
       - active -> is_active=true, archived=false  (DB trigger will enforce readiness)
       - archived -> archived=true, is_active=false
     - On violation from DB trigger, return 400 with
       { code: 'TEMPLATE_NOT_READY', message: 'At least one stage is required.' }

Frontend - Template Detail header (React, shadcn)
- Replace the status pill with a Select. If user lacks permission, render read-only pill.

Example component:
```tsx
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle, } from "@/components/ui/alert-dialog";
import { useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";

type Status = "draft" | "active" | "archived";

export function TemplateStatusControl({ templateId, value, canEdit }:{ templateId:string; value:Status; canEdit:boolean }) {
  const qc = useQueryClient();
  const [dialogOpen, setDialogOpen] = useState(false);
  const [dialogMsg, setDialogMsg] = useState("");

  const { data: readiness } = useQuery({
    queryKey: ["templateReadiness", templateId],
    queryFn: async () => {
      const r = await fetch(`/api/templates/${templateId}/readiness`);
      if (!r.ok) return { active_stage_count: 0 };
      return r.json();
    }
  });

  const mutate = useMutation({
    mutationFn: async (next: Status) => {
      const r = await fetch(`/api/templates/${templateId}/status`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ status: next }),
      });
      if (!r.ok) {
        const err = await r.json().catch(() => ({}));
        throw err;
      }
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["templates"] });
      qc.invalidateQueries({ queryKey: ["template", templateId] });
      qc.invalidateQueries({ queryKey: ["templateReadiness", templateId] });
      toast.success("Template status updated");
    },
    onError: (err:any) => {
      const msg = err?.code === "TEMPLATE_NOT_READY"
        ? "Cannot set to Active. Add at least one stage to this template."
        : (err?.message || "Failed to update status");
      setDialogMsg(msg);
      setDialogOpen(true);
    },
  });

  const onChange = async (next: Status) => {
    if (!canEdit) return;
    // Pre-check for Active to avoid a round trip when possible
    if (next === "active" && (readiness?.active_stage_count ?? 0) < 1) {
      setDialogMsg("Cannot set to Active. Add at least one stage to this template.");
      setDialogOpen(true);
      return;
    }
    mutate.mutate(next);
  };

  return (
    <>
      <Select value={value} onValueChange={onChange} disabled={!canEdit || mutate.isPending}>
        <SelectTrigger className="w-36">
          <SelectValue placeholder="Status" />
        </SelectTrigger>
        <SelectContent>
          <SelectItem value="draft">Draft</SelectItem>
          <SelectItem value="active">Active</SelectItem>
          <SelectItem value="archived">Archived</SelectItem>
        </SelectContent>
      </Select>

      <AlertDialog open={dialogOpen} onOpenChange={setDialogOpen}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Cannot change status</AlertDialogTitle>
            <AlertDialogDescription>{dialogMsg}</AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel type="button">Close</AlertDialogCancel>
            <AlertDialogAction
              onClick={() => {
                setDialogOpen(false);
                const el = document.getElementById("template-stages-section");
                if (el) el.scrollIntoView({ behavior: "smooth", block: "start" });
              }}
            >
              Add Stage
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </>
  );
}

UI notes
	•	Put an id="template-stages-section" on the Stages panel so the dialog CTA can scroll there.
	•	On list view, keep the old pill. Make the dropdown on the detail page only.
	•	Disable the control for users without permission.

Acceptance
	•	Selecting Active with zero active stages shows the dialog and does not change the status.
	•	Selecting Active with at least one stage succeeds.
	•	Selecting Draft or Archived always succeeds.
	•	Server rejects invalid Active status even if the client bypasses checks.
	•	Templates remain independent of candidates.
