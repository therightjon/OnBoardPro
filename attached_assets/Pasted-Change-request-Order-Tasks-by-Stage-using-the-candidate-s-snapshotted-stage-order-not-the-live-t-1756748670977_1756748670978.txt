Change request: Order “Tasks by Stage” using the candidate’s snapshotted stage order (not the live template)

Context
- Candidates cannot be created without a template.
- expand_template runs at creation and produces an independent snapshot of tasks.
- Templates can change later and must NOT affect existing candidates.
- Therefore the stage sequence shown on a candidate must come from a snapshot taken during expand_template.

Goal
- On /candidates/[id], render stage sections in the order captured at template expansion time.
- Do NOT depend on live template_stages after creation.

DB (idempotent)

-- Snapshot table to store the per-candidate stage sequence captured at expand time
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.tables WHERE table_name = 'candidate_template_stages'
  ) THEN
    CREATE TABLE candidate_template_stages (
      id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
      candidate_id uuid NOT NULL REFERENCES candidates(id) ON DELETE CASCADE,
      stage_id uuid NOT NULL REFERENCES hiring_stages(id),
      stage_name_snapshot text NOT NULL,      -- captures the label as it was
      order_index integer NOT NULL,           -- 1..N as captured from the template
      created_at timestamp NOT NULL DEFAULT now(),
      updated_at timestamp NOT NULL DEFAULT now()
    );
    CREATE UNIQUE INDEX uniq_candidate_stage ON candidate_template_stages(candidate_id, stage_id);
    CREATE INDEX idx_candidate_stage_order ON candidate_template_stages(candidate_id, order_index);
  END IF;
END$$;

-- Optional: keep order on each candidate_task row (helps avoid extra joins)
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_name='candidate_tasks' AND column_name='stage_order_index'
  ) THEN
    ALTER TABLE candidate_tasks ADD COLUMN stage_order_index integer;
  END IF;
END$$;

-- Updated_at trigger (if you don't already have the helper)
CREATE OR REPLACE FUNCTION set_updated_at() RETURNS trigger AS $$
BEGIN NEW.updated_at = now(); RETURN NEW; END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_candidate_template_stages_updated_at ON candidate_template_stages;
CREATE TRIGGER trg_candidate_template_stages_updated_at
  BEFORE UPDATE ON candidate_template_stages
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

Server: expand_template(template_id, candidate_id)
- After copying template_tasks → candidate_tasks, also snapshot the stage sequence:

Pseudo-SQL (do inside the same transaction):
-- 1) Read the template’s stages in order at this moment
WITH ts AS (
  SELECT ts.stage_id, hs.name AS stage_name, ts.order_index
  FROM template_stages ts
  JOIN hiring_stages hs ON hs.id = ts.stage_id
  WHERE ts.template_id = $1 AND ts.is_active = TRUE
  ORDER BY ts.order_index ASC
)
-- 2) Write snapshot rows (UPSERT in case of reruns)
INSERT INTO candidate_template_stages (candidate_id, stage_id, stage_name_snapshot, order_index)
SELECT $2, stage_id, stage_name, order_index FROM ts
ON CONFLICT (candidate_id, stage_id)
DO UPDATE SET
  stage_name_snapshot = EXCLUDED.stage_name_snapshot,
  order_index         = EXCLUDED.order_index,
  updated_at          = now();

-- 3) Stamp each candidate_task with its stage order (optional but recommended)
UPDATE candidate_tasks ct
SET stage_order_index = s.order_index
FROM candidate_template_stages s
WHERE s.candidate_id = ct.candidate_id
  AND s.stage_id = ct.stage_id
  AND ct.candidate_id = $2;

-- 4) If you store candidates.current_stage_id, set it to the first snapshot stage
UPDATE candidates c
SET current_stage_id = s.stage_id
FROM candidate_template_stages s
WHERE c.id = $2 AND s.candidate_id = $2 AND s.order_index = 1;

APIs

-- A) Candidate’s ordered stages (from snapshot)
GET /api/candidates/:id/stages
SELECT stage_id, stage_name_snapshot AS stage_name, order_index
FROM candidate_template_stages
WHERE candidate_id = $1
ORDER BY order_index ASC;

-- B) Candidate tasks should include stage_order_index (or you can join it server-side)
GET /api/candidates/:id/tasks
SELECT
  ct.*,
  s.order_index AS stage_order_index
FROM candidate_tasks ct
LEFT JOIN candidate_template_stages s
  ON s.candidate_id = ct.candidate_id AND s.stage_id = ct.stage_id
WHERE ct.candidate_id = $1 AND ct.archived = FALSE;

Frontend (/candidates/[id])

1) Fetch both:
   - /api/candidates/:id/stages  → ordered snapshot
   - /api/candidates/:id/tasks   → tasks with stage_order_index

2) Build order map from the snapshot:
   const orderMap = useMemo(() => {
     const m = new Map<string, number>();
     (stages ?? []).forEach(s => m.set(s.stage_id, s.order_index));
     return m;
   }, [stages]);

3) Group + sort using the snapshot order:
   // attach order for stable sorting
   const withOrder = useMemo(
     () => (tasks ?? []).map(t => ({
       ...t,
       stage_order_index: t.stage_order_index ?? orderMap.get(t.stage_id) ?? Number.MAX_SAFE_INTEGER
     })),
     [tasks, orderMap]
   );

   // group by stage_id, label from snapshot
   const groups = useMemo(() => {
     const by = new Map<string, { name: string; order: number; items: any[] }>();
     for (const t of withOrder) {
       const key = t.stage_id ?? 'none';
       const order = t.stage_order_index;
       const name =
         stages?.find(s => s.stage_id === key)?.stage_name ??
         t.stage?.name ?? 'Unassigned';
       if (!by.has(key)) by.set(key, { name, order, items: [] });
       by.get(key)!.items.push(t);
     }
     return [...by.values()].sort((a, b) => a.order - b.order);
   }, [withOrder, stages]);

4) Render sections in that order; sort items inside by due_at ascending.

Stage advancement (if implemented)
- Use candidate_template_stages order when moving to the next stage (not hiring_stages or template_stages).
- When all required tasks in current stage are done, advance to the snapshot’s next order_index.
- Write to candidate_stage_history as you already do.

Acceptance
- A newly created candidate shows stages in the exact order that existed on the template at creation time.
- Changing the template later does not change any existing candidate’s stage order or labels.
- Deleting/re-adding stages on the template does not affect existing candidates.
- Tasks by Stage groups and the “current stage” value are consistent with the snapshot.
